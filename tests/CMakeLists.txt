  )
  #set(MEMORYCHECK_COMMAND_OPTIONS "--leak-check=full --show-leak-kinds=all --exit-on-first-error=yes --error-exitcode=1")

  set(MEMORYCHECK_COMMAND "")
  set(MEMORYCHECK_COMMAND_OPTIONS "")
  
endif()



function(add_custom_test TEST_NAME)
  add_executable(${TEST_NAME} ${ARGN})
  target_link_libraries(${TEST_NAME}
    PRIVATE
    tld-dev
  )
  # message(STATUS "Running test: ${TEST_NAME} with command: ${MEMORYCHECK_COMMAND} ${MEMORYCHECK_COMMAND_OPTIONS} ${TEST_NAME}")
  add_test(
    NAME ${TEST_NAME}
    COMMAND ${MEMORYCHECK_COMMAND} ${MEMORYCHECK_COMMAND_OPTIONS} ${CMAKE_BINARY_DIR}/tests/${TEST_NAME}
  )
endfunction()

# Single-header build: ./tld.h (generated by tools/amalgamate.py) used without tld-dev,
# in two translation units so duplicate or missing symbols fail to link.
add_executable(single-header single/single_impl.c single/single_use.c)
target_include_directories(single-header PRIVATE ${CMAKE_SOURCE_DIR})
set_target_properties(single-header PROPERTIES C_STANDARD 11 C_EXTENSIONS OFF)
target_link_libraries(single-header PRIVATE m Threads::Threads)
add_test(NAME single-header COMMAND single-header)
# The same two TUs in strict ISO C with warnings as errors: on glibc this caught
# realpath/localtime_r/strnlen being used without declarations.
add_test(NAME single-header-strict
  COMMAND ${CMAKE_C_COMPILER} -std=c11 -Wall -Wextra -pedantic -Werror -I${CMAKE_SOURCE_DIR}
          ${CMAKE_CURRENT_SOURCE_DIR}/single/single_impl.c ${CMAKE_CURRENT_SOURCE_DIR}/single/single_use.c
          -o ${CMAKE_CURRENT_BINARY_DIR}/single-header-strict -lm -lpthread)
